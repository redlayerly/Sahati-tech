[term rnn]

[term gru]

simple = ChatGPT ve Claude gibi dil modelleri, özünde şaşırtıcı derecede basit bir fikirle çalışır: sonraki kelimeyi tahmin etmek. Metni küçük parçalara böler, "dikkat" adı verilen bir mekanizmayla önceki kelimelerden hangilerinin şu anki kelime için önemli olduğuna karar verir ve en olası parçayı seçer. analogy = Cümle tamamlama oyunu gibi: "Fırına ... almaya gittim." "Ekmek" diye tahmin edersin çünkü buna benzer pek çok cümle okudun ve duydun. Model de aynısını yapar, ama milyonlarca kitaba denk gelen metni okumuştur.
why = Bu modeller artık her yerde: arama, eğitim, müşteri hizmetleri, yazılım. Nasıl çalıştıklarını bilmek, bir cevaba ne zaman güvenip ne zaman doğrulaman gerektiğini ve neden bir bilgiyi tam bir özgüvenle "uydurabildiklerini" (halüsinasyon) anlamanı sağlar.
example = Telefonunun klavyesi yazarken bir sonraki kelimeyi önerdiğinde, bu aynı prensiple çalışan çok küçük bir dil modelidir.

[term 05.0]
name = Tokenizasyon
def = Ham metni Transformer modellerinin anlayabileceği ayrık token kimliklerine ayırma işlemi.

[term 05.1]
name = Token
def = Dil modellerinin işlediği ve ürettiği en küçük birimler: alt kelimeler, karakterler veya kelimeler.

[term 05.2]
name = Sözcük Dağarcığı
def = Bir tokenizer içinde her birine benzersiz bir sayı verilmiş izin verilen tokenların sabit sözlüğü.

[term 05.3]
name = BPE
def = Bayt Çifti Kodlaması: sık geçen karakter çiftlerini tekrar tekrar birleştirerek en uygun tokenları oluşturan algoritma.

[term 05.4]
name = WordPiece
def = BERT'in kullandığı, token seçimlerini dil modeli olasılığına göre optimize eden alt kelime yöntemi.

[term 05.5]
name = SentencePiece
def = Metni ham bayt dizisi olarak işleyen, boşluksuz yazılan diller dahil pek çok dili doğrudan destekleyen tokenizer.

[term 05.6]
name = Tokenizer Eğitimi
def = Temsili bir metin derleminden sözcük dağarcığını ve birleştirme kurallarını oluşturan ön hesaplama.

[term 05.7]
name = Gömmeler (Embeddings)
def = Anlamın sayısal vektörlerle temsil edildiği, anlamca yakın kelimelerin birbirine yakın düştüğü uzay.

[term 05.8]
name = Konumsal Kodlama
def = Dikkat katmanlarına kelimelerin sırasını bildirmek için eklenen matematiksel vektörler.

[term 05.9]
name = RoPE
def = Dönen Konum Gömmesi: token konumlarını vektör uzayında döndürme yoluyla kodlar; Llama gibi modellerde yaygındır.

[term 05.10]
name = ALiBi
def = Doğrusal Yanlılıklı Dikkat: dikkat puanlarına mesafeye bağlı ceza uygular; daha uzun metinlere genellemeyi kolaylaştırır.

[term 05.11]
name = Dikkat Mekanizması
def = Bir dizideki tokenlar arasında dinamik olarak ağırlıklı önem puanları hesaplayan mekanizma.

[term 05.12]
name = Öz-Dikkat
def = Metin içindeki ilişkileri yakalamak için aynı dizideki tüm token çiftleri arasında dikkat hesaplamak.

[term 05.13]
name = Çapraz Dikkat
def = Sorguların bir diziden, anahtar ve değerlerin başka bir diziden geldiği dikkat türü; çeviride standarttır.

[term 05.14]
name = Çok Başlı Dikkat

[term 05.15]
name = Nedensel Dikkat

[term 05.16]
name = Maskeleme
def = Gelecekteki tokenları gizlemek veya maskeli dil modellerini eğitmek için dikkat matrisindeki değerleri bilerek kapatmak.

[term 05.17]
name = İleri Beslemeli Ağ

[term 05.18]
name = Artık Bağlantılar
def = Bloğun girdisini doğrudan çıktısına ekleyen kısa yollar; derin ağlarda gradyanların kaybolmasını önler.

[term 05.19]
name = Katman Normalizasyonu

[term 05.20]

[term 05.21]

[term 05.22]
name = Transformer Mimarisi

[pkg 06]
name = Büyük Dil Modelleri
desc = Milyarlarca parametreli temel modellerin ön eğitimi, hizalanması, ölçeklenmesi ve metin üretimi sırasında örneklenmesi.
simple = Büyük bir dil modeli iki aşamadan geçer. Muazzam miktarda metin üzerinde ön eğitim ona dili ve genel bilgiyi öğretir. Ardından sohbet örnekleri ve insan değerlendirmeleriyle yapılan ince ayar onu yararlı ve kibar bir asistana dönüştürür. Kullanırken cevaplarını temperature gibi ayarlarla ve isteğini nasıl yazdığınla (prompt mühendisliği) şekillendirirsin.

example = Bir asistana "adım adım düşün" dediğinde matematik sorusundaki cevabı iyileşiyorsa, düşünce zinciri (Chain of Thought) tekniğini kullanıyorsun demektir.

[term 06.0]
name = Temel Modeller

[term 06.1]

[term 06.2]
name = İnce Ayar

[term 06.3]

[term 06.4]

def = İnsan Geri Bildiriminden Pekiştirmeli Öğrenme: model çıktılarını bir ödül modeli üzerinden insan niyeti ve değerleriyle hizalamak.

[term 06.5]

[term 06.6]

[term 06.7]

[term 06.9]

[term 06.10]
name = Ortaya Çıkan Yetenekler

[term 06.11]

[term 06.12]

[term 06.13]

[term 06.14]

def = Olasılıkları ölçekleyen ayar: düşük değerler daha tutarlı ve kesin, yüksek değerler daha yaratıcı ve çeşitli cevaplar verir.

[term 06.15]

[term 06.16]

[term 06.17]

[term language-models]

[term sampling]

[term prompt-engineering]

[term prompt-templates]

[term system-prompt]

[term few-shot]

[term cot]

[term tool-calling]
name = Araç / Fonksiyon Çağırma

[term structured-output]

[pkg 07]
name = Bağlam Mühendisliği ve RAG Sistemleri
desc = Modelin sınırlı çalışma belleğini yönetmek ve onu arama ile geri getirme yoluyla kaynaklarına bağlamak; böylece güncel ve güvenilir bilgiden cevap verir. Sohbetler arasında hatırlamasını sağlayan bellek sistemleri de buna dahil.
simple = Modelin "bağlam penceresi" denen sınırlı bir çalışma belleği vardır ve şirketinin belgeleri ya da eğitiminden sonra olan hiçbir şey hakkında bilgisi yoktur. Çözüm RAG'dir: kaynaklarında soruyla ilgili pasajları arar, onları soruyla birlikte modelin önüne koyarsın ve model hafızasından değil bunlardan cevap verir. Bu pencereye neyin girip çıkacağını yönetmek artık kendi başına bir uzmanlık: bağlam mühendisliği.
analogy = Açık kitap sınavı. Öğrencinin her şeyi ezberlemesi gerekmez, ama doğru sayfayı hızla bulmayı bilmesi gerekir. Anlamsal arama dizindir, model de öğrencidir.
example = Bir telekom şirketinin destek botu, güncel paket fiyatlarını eğitim sırasında öğrendiği eski bilgilerden değil, şirketin güncel belgelerinden cevaplar.

[term 07.0]
name = Belge Parçalama Stratejileri
def = Uzun belgeleri vektör aramasına ve bağlam sınırına uygun anlamlı parçalara bölmek.

[term 07.1]
name = Yoğun Geri Getirme

[term 07.2]

[term 07.3]
name = Yeniden Sıralama
def = Getirilen adayları ikinci bir sinir modeliyle yeniden puanlayıp en ilgili parçaları öne almak.

[term 07.4]
name = Graph RAG
def = Belgeleri varlıklar ve ilişkilerden oluşan bir bilgi grafiğine dönüştürmek; böylece model çok adımlı karmaşık soruları cevaplayabilir.

[term context-window]
name = Bağlam Penceresi
def = Modelin bir kerede "görebildiği" en fazla metin miktarı (token cinsinden); talimatlar, sohbet, belgeler ve kendi cevabı dahil.

[term context-management]
name = Bağlam Yönetimi

[term context-collapse]

[term context-compression]
name = Bağlam Sıkıştırma

[term context-engineering]
name = Bağlam Mühendisliği
def = Model cevaplamadan önce ona verilen her şeyi tasarlamak: talimatlar, getirilen belgeler, bellek ve araç sonuçları, bilinçli bir sıra ve miktarda.

[term rag]
name = RAG (Geri Getirmeyle Güçlendirilmiş Üretim)
def = Soruyla ilgili bilgiyi harici kaynaklarda arayıp modelin önüne koymak ve oradan cevap vermesini sağlamak; halüsinasyonu azaltır, bilgiyi güncel tutar.

[term semantic-search]
name = Anlamsal Arama
def = Kelime eşleşmesine değil anlama göre aramak; "hesabıma nasıl geri girerim" sorusu "Şifre kurtarma" başlıklı belgeyi bulur.

[term recursive-chunking]
name = Özyinelemeli Parçalama
def = Belgeyi doğal yapısına göre (bölümler, sonra paragraflar, sonra cümleler) her parça doğru boyuta gelene kadar bölmek; bir fikir ortadan kesilmez.

[term prompt-compression]
name = Prompt Sıkıştırma
def = Anlamı koruyarak sonucu etkilemeyen kelimeleri çıkarıp isteği kısaltmak; maliyet düşer, cevap hızlanır.

[term query-expansion]
name = Sorgu Genişletme
def = Arama öncesinde kullanıcının sorusunu yeniden ifade etmek veya eş anlamlılar eklemek; doğru belgeleri bulma ihtimali artar.

[term contextual-retrieval]
name = Bağlamsal Geri Getirme

[term parent-doc]
name = Üst Belge Geri Getirme
def = Küçük ve hassas parçalarla arayıp modele bu parçaların ait olduğu daha büyük paragrafı veya bölümü vermek; model tüm bağlamı görür.

[term metadata-filtering]

[term knowledge-base]

[term memory-systems]

[term short-term-memory]

[term long-term-memory]

[term conversation-memory]
name = Sohbet Belleği
def = Sohbet geçmişini ya da özetini tutmak; böylece model "az önce dediğin gibi" gibi göndermeleri anlar.
